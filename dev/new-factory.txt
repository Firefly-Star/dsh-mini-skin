				const scrimColor = `rgba(${CANVAS_RGB[mode].join(", ")}, ${sidebarAlpha.toFixed(3)})`;
				body.style.setProperty(SIDEBAR_SCRIM_VAR, `linear-gradient(${scrimColor}, ${scrimColor})`);
				if (sidebarChoice === "custom") {
					const url = String(active.sidebarImageUrl || "").trim();
					if (url === "") body.style.removeProperty(SIDEBAR_CUSTOM_VAR);
					else body.style.setProperty(SIDEBAR_CUSTOM_VAR, `url("${url}")`);
				}
				if (sidebarChoice === "file") {
					const url = fileUrls[`sidebar:${mode}`];
					if (url === null) body.style.removeProperty(SIDEBAR_FILE_VAR);
					else body.style.setProperty(SIDEBAR_FILE_VAR, `url("${url}")`);
				} else {
					body.style.removeProperty(SIDEBAR_FILE_VAR);
				}
			}

			// 字体
			const uiStack = FONT_STACKS[settings.uiFont];
			if (uiStack === undefined) body.style.removeProperty("--dsw-font-family");
			else body.style.setProperty("--dsw-font-family", uiStack);
			const sidebarStack = FONT_STACKS[settings.sidebarFont];
			if (sidebarStack === undefined) {
				body.removeAttribute("data-mini-sidebar-font");
				body.style.removeProperty(SIDEBAR_FONT_VAR);
			} else {
				body.setAttribute("data-mini-sidebar-font", "");
				body.style.setProperty(SIDEBAR_FONT_VAR, sidebarStack);
			}

			// 造型开关
			body.toggleAttribute("data-mini-square", settings.square === true);
			body.toggleAttribute("data-mini-chip", settings.linkChip === true);
			body.setAttribute("data-mini-brand", settings.brandMark === "host" ? "host" : "dsh");
		};

			for (const name of [CANVAS_CUSTOM_VAR, CANVAS_FILE_VAR, SIDEBAR_CUSTOM_VAR, SIDEBAR_FILE_VAR, CANVAS_TOKEN, SIDEBAR_SCRIM_VAR, SIDEBAR_FONT_VAR, "--dsh-mini-skin-sidebar-x", "--dsh-mini-skin-sidebar-y", "--dsw-font-family"]) {

		const fileUrls = { "canvas:dark": null, "canvas:light": null, "sidebar:dark": null, "sidebar:light": null };

		const openImageDatabase = () =>
			new Promise((resolve, reject) => {
				const request = window.indexedDB.open("dsh-mini-skin", 1);
				request.onupgradeneeded = () => {
					if (!request.result.objectStoreNames.contains("images")) request.result.createObjectStore("images");
				};
				request.onsuccess = () => resolve(request.result);
				request.onerror = () => reject(request.error);
			});

		const storeImage = async (key, blob) => {
			const db = await openImageDatabase();
			await new Promise((resolve, reject) => {
				const tx = db.transaction("images", "readwrite");
				tx.objectStore("images").put(blob, key);
				tx.oncomplete = () => resolve();
				tx.onerror = () => reject(tx.error);
			});
			db.close();
		};

		const readImage = async (key) => {
			const db = await openImageDatabase();
			const blob = await new Promise((resolve, reject) => {
				const tx = db.transaction("images", "readonly");
				const request = tx.objectStore("images").get(key);
				request.onsuccess = () => resolve(request.result === undefined ? null : request.result);
				request.onerror = () => reject(request.error);
			});
			db.close();
			return blob;
		};

					return await handles[0].getFile();

				if (error !== null && error.name === "AbortError") return null; // 用户取消
			}
			return await new Promise((resolve) => {
				const input = document.createElement("input");
				input.type = "file";
				input.accept = "image/*";

			for (const slot of ART_SLOTS) {
				const [area, mode] = slot.split(":");

				try { blob = await readImage(slot); } catch { blob = null; }

				if (key === "activeSkin") continue;
				shared[key] = settings[key];

			return {
				pack: {
					format: PACK_FORMAT,
					version: PACK_VERSION,
					name: settings.packName === "" ? "dsh-mini-skin" : settings.packName,
					savedAt: new Date().toISOString(),
					shared,
					modes: { dark: { ...settings.dark }, light: { ...settings.light } },
					images,
				},
				count: Object.keys(images).length,
				bytes,

		};

			downloadText(`${safe}-${new Date().toISOString().slice(0, 10)}.json`, JSON.stringify(pack, null, 1));
			return `已导出「${pack.name}」：内置图按名引用，另内嵌 ${count} 张本地图（${(bytes / 1048576).toFixed(2)} MB）`;

				...(parsed.shared !== null && typeof parsed.shared === "object" ? parsed.shared : {}),
				dark: modes.dark,
				light: modes.light,
				packName: typeof parsed.name === "string" ? parsed.name : "",
			});
			return { settings: next, message: `已导入「${next.packName === "" ? "未命名" : next.packName}」：恢复 ${restored} 张本地图` };
		};

		/** 与拉条双向同步的数值框：输入过程中不提交，失焦或回车才夹到范围并写回。 */
		function NumberBox(props) {
			const [draft, setDraft] = React.useState(String(props.value));
			React.useEffect(() => {
				setDraft(String(props.value));
			}, [props.value]);
			const commit = (raw) => {
				// 清空输入框后失焦视为"放弃修改"，不要把它当成 0。
				if (String(raw).trim() === "") {
					setDraft(String(props.value));
					return;
				}
				const parsed = Number(raw);
				const next = Number.isFinite(parsed) ? Math.min(props.max, Math.max(props.min, Math.round(parsed))) : props.value;
				setDraft(String(next));
				props.onCommit(next);
			};
			return React.createElement(
				React.Fragment,
				null,
				React.createElement("input", {
					type: "number",
					min: props.min,
					max: props.max,
					step: 1,
					value: draft,
					"aria-label": props.label,
					style: NUMBER_STYLE,
					onChange: (event) => setDraft(event.currentTarget.value),
					onBlur: (event) => commit(event.currentTarget.value),
					onKeyDown: (event) => {
						if (event.key === "Enter") {
							commit(event.currentTarget.value);
							event.currentTarget.blur();
						}
					},
				}),
				props.unit === undefined || props.unit === null ? null : React.createElement("span", { style: UNIT_STYLE }, props.unit),
			);
		}

		/** 「自定义皮肤」分区。只有设置弹窗打开时才挂载；关掉即卸载，平时零成本。 */
		function CustomSkinPanel(props) {
			const [settings, setSettings] = React.useState(() => (typeof props.read === "function" ? props.read() : props.settings));
			// 提交放在事件处理器里，不放进 setState 的更新函数：更新函数里做副作用会被
			// React 重放或丢弃，正是"改了控件没反应"的常见来源。
			const change = (patch) => {
				const next = { ...settings, ...patch };
				setSettings(next);
				props.commit(next);
			};

			// 面板一次编辑一套预设：默认跟随当前 DSH 主题，也可手动切去看另一套。
			const [editingMode, setEditingMode] = React.useState(document.body !== null && document.body.hasAttribute("data-ds-dark-theme") ? "dark" : "light");
			const m = settings[editingMode];
			const patchMode = (patch) => {
				// 改哪一套就让哪一套生效：setTheme 在偏好未变时会直接返回，所以拖滑块也不会重复切换。
				if (typeof props.switchMode === "function") props.switchMode(editingMode);
				change({ [editingMode]: { ...m, ...patch } });
			};
			const [packStatus, setPackStatus] = React.useState("");
			const [library, setLibrary] = React.useState([]);
			const [libraryDir, setLibraryDir] = React.useState("");
			const [libraryError, setLibraryError] = React.useState("");

			const refreshLibrary = async () => {
				try {
					const payload = await fetchLibrary();
					setLibrary(payload.skins);
					setLibraryDir(payload.dir);
					setLibraryError("");
				} catch (error) {
					setLibrary([]);
					setLibraryError(error !== null && error !== undefined && error.message !== undefined ? error.message : String(error));
				}
			};
			React.useEffect(() => {
				void refreshLibrary();
			}, []);

			const currentHash = fingerprint(settings);
			const activeSkin = settings.activeSkin !== null && typeof settings.activeSkin === "object" ? settings.activeSkin : { kind: "custom" };
			const skinIsNamed = activeSkin.kind === "pack" || activeSkin.kind === "builtin";
			const skinOnPack = skinIsNamed && activeSkin.hash === currentHash;
			const skinStatus = activeSkin.kind === "official" ? "官方皮肤" : skinOnPack ? activeSkin.name : "自定义";
			const skinValue = activeSkin.kind === "official" ? "official" : skinOnPack ? (activeSkin.kind === "builtin" ? "builtin" : `pack:${activeSkin.file}`) : "custom";

			const applyLibrarySkin = async (value) => {
				if (value === "custom") return;
				if (value === "builtin") {
					// 内置皮肤 = 出厂那套：深色→暗色场景、浅色→亮色场景、侧栏→立绘。
					const next = buildSettings(null);
					next.activeSkin = { kind: "builtin", name: "内置皮肤", hash: fingerprint(next) };
					setSettings(next);
					props.commit(next);
					setPackStatus("已切换到内置皮肤（场景 + 立绘，深/浅各一套）。");
					return;
				}
				if (value === "official") {
					const next = { ...settings, activeSkin: { kind: "official" } };
					setSettings(next);
					props.commit(next);
					setPackStatus("已切换到官方皮肤（宿主原样）。");
					return;
				}
				const file = value.startsWith("pack:") ? value.slice(5) : value;
				setPackStatus("正在应用…");
				try {
					const pack = await readLibrarySkin(file);
					const result = await applySkinPack(pack, { file });
					setSettings(result.settings);
					props.commit(result.settings);
					setPackStatus(`已应用「${result.settings.packName === "" ? file : result.settings.packName}」：恢复 ${result.restored} 张本地图`);
				} catch (error) {
					setPackStatus(`应用失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`);
				}
			};

			const runExport = async () => {
				setPackStatus("正在导出…");
				try {
					const message = await exportPack(settings);
					let extra = "";
					try {
						const built = await buildPack(settings);
						const saved = await saveToLibrary(built.pack);
						extra = `；已存入皮肤库（${saved.file}）`;
						await refreshLibrary();
					} catch (error) {
						extra = `；存入皮肤库失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`;
					}
					setPackStatus(message + extra);
				} catch (error) {
					setPackStatus(`导出失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`);
				}
			};

			const runImport = async () => {
				setPackStatus("正在导入…");
				try {

				} catch (error) {

				}
			};

							...(skinValue === "custom" ? [{ value: "custom", label: "自定义（当前设置）" }] : []),
						],

					),

					Select(editingMode, [{ value: "dark", label: "深色模式预设" }, { value: "light", label: "浅色模式预设" }], (value) => {
						setEditingMode(value);

					}),

		// ===================== 装载 =====================
		function apply(ctx) {
			const claimant = Symbol("dsh-mini-skin-activation");
			if (!activate(claimant)) document.addEventListener("DOMContentLoaded", () => activate(claimant), { once: true });

			let settings = readSettings();
			applySettings(claimant, settings);

			// 本地图片本体存在 IndexedDB 里；取出来生成 blob URL 之后重放一次设置。
			void (async () => {
				for (const slot of ART_SLOTS) {
					try {
						// 旧版只有 canvas / sidebar 两个槽：迁移时两套预设都指向同一张图。
						const blob = (await readImage(slot)) ?? (slot.endsWith(":dark") ? await readImage(slot.split(":")[0]) : null);
						if (blob !== null) fileUrls[slot] = URL.createObjectURL(blob);
					} catch {
						/* 读不到就当作没有本地图 */
					}
				}
				if (document.body !== null) applySettings(claimant, settings);
			})();

			// 切浅/深色时换成对应预设。色值取自事件快照（active.colorScheme），绝不读 DOM；
			// 订阅走 ctx.inject，避免 apply 早于 theme 服务注册时静默漏订阅。
			const resync = (snapshot) => {
				const active = snapshot !== null && typeof snapshot === "object" ? snapshot.active : undefined;
				const scheme = active !== null && typeof active === "object" ? active.colorScheme : undefined;
				if (scheme === "dark" || scheme === "light") knownScheme = scheme;
				applySettings(claimant, settings);
			};
			const subscribeTheme = (themeCtx) => {
				const theme = typeof themeCtx.get === "function" ? themeCtx.get("theme") : undefined;
				if (theme === undefined || typeof themeCtx.on !== "function") return;
				themeCtx.on("theme/change", resync);
				themeCtx.effect(() => () => themeCtx.off("theme/change", resync), "dsh-mini-skin: theme resync");
			};
			if (typeof ctx.inject === "function") ctx.inject(["theme"], subscribeTheme);
			else subscribeTheme(ctx);

			// 设置分区：用 ctx.inject 等 slots 服务就绪再注册（本插件 dsh.client.inject 为空，
			// apply 可能早于 slots 注册，查一次 ctx.get 会拿到 undefined）。
			const registerSection = (slotsCtx) => {
				const slots = typeof slotsCtx.get === "function" ? slotsCtx.get("slots") : undefined;
				if (slots === undefined || typeof slots.inject !== "function") return;
				slotsCtx.effect(
					() =>
						slots.inject("settings.section", () =>
							slots.register(
								{
									name: "settings.section",
									id: SECTION_ID,
									order: 116,
									label: () => SECTION_TITLE,
									inject: () => ({
											// 面板每次挂载都通过它取当前值：不能传对象快照，否则重新挂载后面板会显示旧值
											// （曾经因此出现"下拉显示旧选项、再选同一项因值未变而不触发"的问题）。
											read: () => settings,
										settings,
										commit: (next) => {
											settings = next;
											persistSettings(next);
											applySettings(claimant, next);
										},
										switchMode: (mode) => {
											// 宿主的唯一偏好写入口：真的切主题（并持久化），这样配另一套预设时能立刻看到效果。
											// 先记账再切换：setTheme 之后宿主要到下一帧才把主题属性写进 DOM。
											if (mode === "dark" || mode === "light") knownScheme = mode;
											const theme = typeof ctx.get === "function" ? ctx.get("theme") : undefined;
											if (theme !== undefined && typeof theme.setTheme === "function") theme.setTheme(mode);
										},
									}),
								},
								CustomSkinPanel,
							),
						),
					"dsh-mini-skin: settings section",
				);
			};
			if (typeof ctx.inject === "function") ctx.inject(["slots"], registerSection);
			else registerSection(ctx);

			ctx.effect(
				() => () => {
					clearSettings(claimant);
					deactivate(claimant);
				},
				"dsh-mini-skin: stylesheet",
			);
		}

		exports.apply = apply;
		// 模块级 inject：客户端运行器会等 slots 服务就绪再调用 apply（皮肤管理器同款做法）。
		// apply 里仍保留 ctx.inject 兜底，两条路径都通。
		exports.inject = ["slots"];
		return module.exports;
	},
