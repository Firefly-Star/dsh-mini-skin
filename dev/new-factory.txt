			shared.bodyClaims.delete(claimant);
			if (shared.bodyClaims.size > 0 || document.body === null) return;
			if (originalBodyValue === null) document.body.removeAttribute(BODY_ATTRIBUTE);
			else document.body.setAttribute(BODY_ATTRIBUTE, originalBodyValue);
		};

		// ===================== 设置：读取 / 保存 / 应用 =====================
		const clamp = (value, min, max) => Math.min(max, Math.max(min, value));
		const isMode = () => (document.body !== null && document.body.hasAttribute("data-ds-dark-theme") ? "dark" : "light");
		// 宿主先发 theme/change、后把主题属性写进 DOM，所以事件里读 body 会读到旧模式。
		// 这里记住"最近一次已知的色值"（事件快照或我们自己的切换），只有完全不知道时才读 DOM。
		let knownScheme = null;
		const currentScheme = () => (knownScheme === "dark" || knownScheme === "light" ? knownScheme : isMode());

		const readSettings = () => {
			let stored = {};
			try {
				const raw = window.localStorage.getItem(SETTINGS_KEY);
				if (raw !== null) {
					const parsed = JSON.parse(raw);
					if (parsed !== null && typeof parsed === "object") stored = parsed;
				} else {
					// 从上一版只有一条不透明度的键迁移过来。必须先判键存在：键不存在时
					// Number(null) 是 0，会被误当成"用户把不透明度调到了 0"，全新安装就看不见图。
					const legacyRaw = window.localStorage.getItem(LEGACY_OPACITY_KEY);
					if (legacyRaw !== null) {
						const legacy = Number(legacyRaw);
						if (Number.isFinite(legacy)) stored = { canvasOpacity: legacy };
					}
				}
			} catch {
				stored = {};
			}
			const merged = buildSettings(stored);
			for (const key of PRESET_KEYS) {
				merged[key].canvasOpacity = clamp(Number(merged[key].canvasOpacity) || 0, 0, 100);
				merged[key].sidebarOpacity = clamp(Number(merged[key].sidebarOpacity) || 0, 0, 100);
			}
			return merged;
		};

		const persistSettings = (settings) => {
			try {
				window.localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
			} catch {
				/* 隐私模式下写不进去 */
			}
		};

		/** 把一份设置完整应用到文档：行内变量 + body 属性。 */
		const applySettings = (claimant, settings) => {
			const body = document.body;
			if (body === null) return;
			// 官方皮肤：撤下本插件的一切（样式表、属性、行内变量），但插件本体仍然活着——设置分区还在。
			if (isOfficial(settings)) {
				clearSettings(claimant);
				deactivate(claimant);
				return;
			}
			activate(claimant);
			shared.settingsClaims.add(claimant);
			const mode = currentScheme();
			const active = settings[presetKey(mode, isWorking())] ?? settings[presetKey(mode, false)];

			// 图片数据一律留在样式表里的自定义属性上，行内只写"选了哪一张"和极小的值：
			// 几百 KB 的 data URI 写进行内样式并不可靠（上一版图不显示的根因）。
			const canvasChoice = CANVAS_CHOICES.some((entry) => entry.value === active.canvasImage) ? active.canvasImage : "none";
			body.setAttribute("data-mini-canvas", canvasChoice);
			if (canvasChoice === "custom") {
				const url = String(active.canvasImageUrl || "").trim();
				writeArtVar(CANVAS_CUSTOM_VAR, url === "" ? null : `url("${url}")`);
			}
			if (canvasChoice === "file") {
				const slot = `canvas:${presetKey(mode, isWorking())}`;
				// 缓存里没有就顺手去取（取到会有回调重放一次）；这一拍先撤下，别留 undefined。
				if (!ART_SLOT_CACHE.has(slot)) void artUrl(slot).then(() => { if (document.body !== null) applySettings(claimant, settings); });
				const url = ART_SLOT_CACHE.get(slot);
				writeArtVar(CANVAS_FILE_VAR, url === null || url === undefined ? null : `url("${url}")`);
			} else {
				// 撤下/换掉背景图时必须清掉这张行内变量，别把上一张图的 blob URL 留在 DOM 上。
				writeArtVar(CANVAS_FILE_VAR, null);
			}

			// 主画布底色 alpha（图越强，底色越透）
			const canvasAlpha = (100 - clamp(active.canvasOpacity, 0, 100)) / 100;
			body.style.setProperty(CANVAS_TOKEN, `rgba(${CANVAS_RGB[mode].join(", ")}, ${canvasAlpha.toFixed(3)})`);

			// 侧栏图的位置微调（px）：默认 0 = 水平居中、底边贴住下方浮雕栏的上沿。

			// 行内只写很小的数值，图片本体始终留在样式表里。

			body.style.setProperty("--dsh-mini-skin-sidebar-x", `${Math.round(Number(active.sidebarOffsetX) || 0)}px`);

			body.style.setProperty("--dsh-mini-skin-sidebar-y", `${Math.round(Number(active.sidebarOffsetY) || 0)}px`);

			// 顶部角色图：图片沿用当前这套的侧栏图，所以空闲放眨眼、工作放敲键盘，自动跟着状态换。
			const characterHeight = Math.max(0, Math.min(320, Math.round(Number(active.characterHeight) || 0)));
			const characterChoice = CHARACTER_CHOICES.some((entry) => entry.value === active.characterImage) ? active.characterImage : "none";
			const characterSlot = `character:${presetKey(mode, isWorking())}`;
			if (characterChoice === "file" && !ART_SLOT_CACHE.has(characterSlot)) {
				void artUrl(characterSlot).then(() => { if (document.body !== null) applySettings(claimant, settings); });
			}
			const characterArt = characterChoice === "file" ? ART_SLOT_CACHE.get(characterSlot) : null;
			if (characterHeight > 0 && characterArt !== null && characterArt !== undefined) {
				body.setAttribute("data-mini-character", "");
				// 别写死 px：侧栏里那块图是固定尺寸的 flex item，写死就会在窗口变矮时和工作区列表
				// 抢空间（工作区先被压没，图最后才缩）。这里让它跟着视口高度走，
				// 上限仍是用户设的值 —— 变矮时图和列表一起缩，至少不会把列表整体挤掉。
				writeArtVar("--dsh-mini-skin-character-h", `clamp(72px, 18vh, ${characterHeight}px)`);
				writeArtVar("--dsh-mini-skin-character-art", `url("${characterArt}")`);
			} else {
				body.removeAttribute("data-mini-character");
				writeArtVar("--dsh-mini-skin-character-h", null);
				writeArtVar("--dsh-mini-skin-character-art", null);
			}


			// 侧栏：同色蒙版 alpha + 选了哪一张（同样不碰层色 token）
			const sidebarChoice = SIDEBAR_CHOICES.some((entry) => entry.value === active.sidebarImage) ? active.sidebarImage : "none";
			const sidebarAlpha = (100 - clamp(active.sidebarOpacity, 0, 100)) / 100;
			if (sidebarChoice === "none") {
				body.removeAttribute("data-mini-sidebar");
				body.removeAttribute("data-mini-sidebar-art");
			} else {
				body.setAttribute("data-mini-sidebar", "");
				body.setAttribute("data-mini-sidebar-art", sidebarChoice);
				// 蒙版必须包成"图像"：rgba() 是颜色，直接放进 background-image 会让整条声明失效
				// （这正是侧栏图不显示、不透明度没反应的原因）。同色 linear-gradient 才是图像。
				const scrimColor = `rgba(${CANVAS_RGB[mode].join(", ")}, ${sidebarAlpha.toFixed(3)})`;
				writeArtVar(SIDEBAR_SCRIM_VAR, `linear-gradient(${scrimColor}, ${scrimColor})`);
				if (sidebarChoice === "custom") {
					const url = String(active.sidebarImageUrl || "").trim();
					writeArtVar(SIDEBAR_CUSTOM_VAR, url === "" ? null : `url("${url}")`);
				}
				if (sidebarChoice === "file") {
					const slot = `sidebar:${presetKey(mode, isWorking())}`;
					if (!ART_SLOT_CACHE.has(slot)) void artUrl(slot).then(() => { if (document.body !== null) applySettings(claimant, settings); });
					const url = ART_SLOT_CACHE.get(slot);
					writeArtVar(SIDEBAR_FILE_VAR, url === null || url === undefined ? null : `url("${url}")`);
				} else {
					writeArtVar(SIDEBAR_FILE_VAR, null);
				}
			}

			// 字体
			const uiStack = FONT_STACKS[settings.uiFont];
			if (uiStack === undefined) body.style.removeProperty("--dsw-font-family");
			else body.style.setProperty("--dsw-font-family", uiStack);
			const sidebarStack = FONT_STACKS[settings.sidebarFont];
			if (sidebarStack === undefined) {
				body.removeAttribute("data-mini-sidebar-font");
				body.style.removeProperty(SIDEBAR_FONT_VAR);
			} else {
				body.setAttribute("data-mini-sidebar-font", "");
				body.style.setProperty(SIDEBAR_FONT_VAR, sidebarStack);
			}

			// 造型开关
			body.toggleAttribute("data-mini-square", settings.square === true);
			body.toggleAttribute("data-mini-chip", settings.linkChip === true);
			body.setAttribute("data-mini-brand", settings.brandMark === "host" ? "host" : "dsh");
		};

		/** 卸载时收回本次激活写入的行内属性；还有别的激活活着就不动。 */
		const clearSettings = (claimant) => {
			shared.settingsClaims.delete(claimant);
			const body = document.body;
			if (body === null || shared.settingsClaims.size > 0) return;
			for (const name of [CANVAS_CUSTOM_VAR, CANVAS_FILE_VAR, SIDEBAR_CUSTOM_VAR, SIDEBAR_FILE_VAR, CANVAS_TOKEN, SIDEBAR_SCRIM_VAR, SIDEBAR_FONT_VAR, "--dsh-mini-skin-sidebar-x", "--dsh-mini-skin-sidebar-y", "--dsh-mini-skin-character-h", "--dsh-mini-skin-character-art", "--dsw-font-family"]) {
				body.style.removeProperty(name);
			}
			// 上面是直接 removeProperty，绕过了"值没变就不写"的缓存 —— 必须一起清掉，
			// 否则下次激活时那次写入会被当成"重复"而跳过，属性就再也不出现了。
			lastArtWritten.clear();
			for (const slot of [...ART_SLOT_CACHE.keys()]) artRevoke(slot);
			for (const name of ["data-mini-canvas", "data-mini-sidebar", "data-mini-sidebar-art", "data-mini-sidebar-font", "data-mini-square", "data-mini-chip", "data-mini-brand", "data-mini-character"]) {
				body.removeAttribute(name);
			}
		};

		// ===================== 本地图片：系统文件选择器 + IndexedDB =====================
		// localStorage 装不下几 MB 的图，所以文件本体存 IndexedDB，内存里只留 blob URL。
		//
		// 缓存键必须就是**图片槽位**（`区域:模式:状态`，见 ART_SLOTS）。曾经这里预置了四个
		// 旧的两段键（canvas:dark…），而读写都用三段键 —— 对象上根本没有那些键，读到的是
		// undefined：`url("undefined")` 让图不显示，`undefined !== null` 又让卸载时
		// revokeObjectURL(undefined) 抛异常、把整段清理中断。所以这里改用 Map + 惰性读取：
		// 读到就缓存，读不到就是"没有这张图"。
		const ART_SLOT_CACHE = new Map();
		/**
		 * 内容指纹 -> blob URL。**这是"切状态不闪"的关键**：皮肤包导入时，同一个字节的图会被写进
		 * 多个槽位（比如女仆皮肤四套预设共用一张场景图）。若每个槽位各建一个 blob URL，
		 * 状态切换就等于"换了一张图"：属性值变了、浏览器重新解码那张大位图，于是闪一下。
		 * 按内容指纹复用同一个 URL，切换时属性值根本不变。
		 */
		const ART_URL_BY_SOURCE = new Map();
		/** SourceBuffer 之类没法读 text() 时退回"大小 + 类型"，同字节的 Blob 一定同大小。 */
		const artFingerprint = async (blob) => {
			try {
				const text = await blob.text();
				return `${blob.size}:${text}`;
			} catch {
				return `${blob.size}:${blob.type}`;
			}
		};

		/** 按内容取（必要时创建）blob URL。 */
		const artUrlForBlob = async (blob) => {
			const fingerprint = await artFingerprint(blob);
			if (ART_URL_BY_SOURCE.has(fingerprint)) return ART_URL_BY_SOURCE.get(fingerprint);
			const url = URL.createObjectURL(blob);
			ART_URL_BY_SOURCE.set(fingerprint, url);
			return url;
		};

		/** 持久化的图片数据 → blob URL（缓存）。读不到返回 null。 */
		const artUrl = async (slot) => {
			if (ART_SLOT_CACHE.has(slot)) return ART_SLOT_CACHE.get(slot);
			let blob = null;
			try {
				const parts = slot.split(":");
				// 旧版只有 canvas / sidebar 两段键（更早还有一段键）：迁移时当作该区域所有预设的图。
				blob = (await readImage(slot)) ?? (await readImage(parts.slice(0, 2).join(":"))) ?? (await readImage(parts[0]));
			} catch {
				/* 隐私模式或读失败：当作没有本地图 */
			}
			const url = blob === null || blob === undefined ? null : await artUrlForBlob(blob);
			ART_SLOT_CACHE.set(slot, url);
			return url;
		};

		/** 存储一张新图（系统选择器或皮肤包导入都走这里），并换掉该槽位旧的缓存 URL。 */
		const artStore = async (slot, blob) => {
			try {
				await storeImage(slot, blob);
			} catch {
				/* 隐私模式或配额不足：本次仍可使用，只是不持久 */
			}
			const previous = ART_SLOT_CACHE.has(slot) ? ART_SLOT_CACHE.get(slot) : null;
			const url = await artUrlForBlob(blob);
			ART_SLOT_CACHE.set(slot, url);
			// 旧 URL 只在"没有别的槽位还在引用它"时才回收 —— 去重之后多个槽位会共享同一个 URL。
			if (previous !== null && previous !== undefined && previous !== url && ![...ART_SLOT_CACHE.values()].some((other) => other === previous)) {
				URL.revokeObjectURL(previous);
				for (const [fingerprint, mapped] of ART_URL_BY_SOURCE) if (mapped === previous) ART_URL_BY_SOURCE.delete(fingerprint);
			}
			forgetArtVar(CANVAS_FILE_VAR);
			forgetArtVar(SIDEBAR_FILE_VAR);
			forgetArtVar("--dsh-mini-skin-character-art");
			return url;
		};

		const artRevoke = (slot) => {
			if (!ART_SLOT_CACHE.has(slot)) return;
			const url = ART_SLOT_CACHE.get(slot);
			ART_SLOT_CACHE.delete(slot);
			// 去重后一个 URL 可能被多个槽位共用：还有别人在用就绝不回收。
			if (url !== null && url !== undefined && ![...ART_SLOT_CACHE.values()].some((other) => other === url)) {
				URL.revokeObjectURL(url);
				for (const [fingerprint, mapped] of ART_URL_BY_SOURCE) if (mapped === url) ART_URL_BY_SOURCE.delete(fingerprint);
				forgetArtVar(CANVAS_FILE_VAR);
				forgetArtVar(SIDEBAR_FILE_VAR);
				forgetArtVar("--dsh-mini-skin-character-art");
			}
		};

		const openImageDatabase = () =>
			new Promise((resolve, reject) => {
				const request = window.indexedDB.open("dsh-mini-skin", 1);
				request.onupgradeneeded = () => {
					if (!request.result.objectStoreNames.contains("images")) request.result.createObjectStore("images");
				};
				request.onsuccess = () => resolve(request.result);
				request.onerror = () => reject(request.error);
			});

		const storeImage = async (key, blob) => {
			const db = await openImageDatabase();
			await new Promise((resolve, reject) => {
				const tx = db.transaction("images", "readwrite");
				tx.objectStore("images").put(blob, key);
				tx.oncomplete = () => resolve();
				tx.onerror = () => reject(tx.error);
			});
			db.close();
		};

		const readImage = async (key) => {
			const db = await openImageDatabase();
			const blob = await new Promise((resolve, reject) => {
				const tx = db.transaction("images", "readonly");
				const request = tx.objectStore("images").get(key);
				request.onsuccess = () => resolve(request.result === undefined ? null : request.result);
				request.onerror = () => reject(request.error);
			});
			db.close();
			return blob;
		};

		/** 系统文件选择器：Chrome 走 showOpenFilePicker（资源管理器那种弹窗），其余退回 <input type=file>。 */
		const pickImageFile = async () => {
			try {
				if (typeof window.showOpenFilePicker === "function") {
					const handles = await window.showOpenFilePicker({
						multiple: false,
						types: [{ description: "图片", accept: { "image/*": [".png", ".jpg", ".jpeg", ".webp", ".gif", ".avif", ".bmp"] } }],
					});
					if (handles.length === 0) return null;
					return await handles[0].getFile();
				}
			} catch (error) {
				if (error !== null && error.name === "AbortError") return null; // 用户取消
			}
			return await new Promise((resolve) => {
				const input = document.createElement("input");
				input.type = "file";
				input.accept = "image/*";
				input.style.display = "none";
				input.addEventListener("change", () => {
					const file = input.files !== null && input.files.length > 0 ? input.files[0] : null;
					input.remove();
					resolve(file);
				}, { once: true });
				input.addEventListener("cancel", () => { input.remove(); resolve(null); }, { once: true });
				document.body.appendChild(input);
				input.click();
			});
		};

		/** 弹选择器并把文件收进 IndexedDB；返回文件名，取消返回 null。 */
		const adoptLocalFile = async (slot) => {
			const file = await pickImageFile();
			if (file === null) return null;
			await artStore(slot, file);
			return file.name;
		};

		// ===================== 皮肤包：导出 / 导入（只涉及本地图片，因此一定自包含）=====================
		const PACK_FORMAT = "dsh-mini-skin-pack";
		const PACK_VERSION = 3;

		const blobToDataUrl = (blob) =>
			new Promise((resolve, reject) => {
				const reader = new FileReader();
				reader.onload = () => resolve(String(reader.result));
				reader.onerror = () => reject(reader.error);
				reader.readAsDataURL(blob);
			});

		const dataUrlToBlob = async (dataUrl) => (await fetch(dataUrl)).blob();

		/** 系统文件选择器（只收 .json）：与选图同一套机制。 */
		const pickPackFile = async () => {
			try {
				if (typeof window.showOpenFilePicker === "function") {
					const handles = await window.showOpenFilePicker({
						multiple: false,
						types: [{ description: "皮肤包", accept: { "application/json": [".json"] } }],
					});
					if (handles.length === 0) return null;
					return await handles[0].getFile();
				}
			} catch (error) {
				if (error !== null && error.name === "AbortError") return null;
			}
			return await new Promise((resolve) => {
				const input = document.createElement("input");
				input.type = "file";
				input.accept = "application/json,.json";
				input.style.display = "none";
				input.addEventListener("change", () => {
					const file = input.files !== null && input.files.length > 0 ? input.files[0] : null;
					input.remove();
					resolve(file);
				}, { once: true });
				input.addEventListener("cancel", () => { input.remove(); resolve(null); }, { once: true });
				document.body.appendChild(input);
				input.click();
			});
		};

		const downloadText = (filename, text) => {
			const url = URL.createObjectURL(new Blob([text], { type: "application/json" }));
			const link = document.createElement("a");
			link.href = url;
			link.download = filename;
			document.body.appendChild(link);
			link.click();
			link.remove();
			window.setTimeout(() => URL.revokeObjectURL(url), 2000);
		};

		/** 收集成包：内置图按名字引用，只把本地图片的字节内嵌进 images。 */
		const buildPack = async (settings) => {
			const images = {};
			let bytes = 0;
			for (const slot of ART_SLOTS) {
				const parts = slot.split(":");
				const area = parts[0];
				const preset = parts[1] + ":" + parts[2];
				const target = settings[preset];
				// 包里一份设置理论上四套俱全；缺的那套跳过，别让一次导出整例炸掉。
				if (target === null || typeof target !== "object") continue;
				const choice = area === "canvas" ? target.canvasImage : area === "character" ? target.characterImage : target.sidebarImage;
				if (choice !== "file") continue;
				let blob = null;
				try { blob = await readImage(slot); } catch { blob = null; }
				if (blob === null || blob === undefined) continue;
				images[slot] = await blobToDataUrl(blob);
				bytes += blob.size;
			}
			const shared = {};
			for (const key of Object.keys(SHARED_DEFAULTS)) {
				// 皮肤库身份不进包：由导入方在导入后自己决定。
				if (key === "activeSkin") continue;
				shared[key] = settings[key];
			}
			return {
				pack: {
					format: PACK_FORMAT,
					version: PACK_VERSION,
					name: settings.packName === "" ? "dsh-mini-skin" : settings.packName,
					savedAt: new Date().toISOString(),
					shared,
					modes: Object.fromEntries(PRESET_KEYS.map((key) => [key, { ...settings[key] }])),
					images,
				},
				count: Object.keys(images).length,
				bytes,
			};
		};

		const exportPack = async (settings) => {
			const { pack, count, bytes } = await buildPack(settings);
			const safe = String(pack.name).replace(/[^\w\u4e00-\u9fa5.-]+/g, "_").slice(0, 48) || "dsh-mini-skin";
			downloadText(`${safe}-${new Date().toISOString().slice(0, 10)}.json`, JSON.stringify(pack, null, 1));
			return `已导出「${pack.name}」：内置图按名引用，另内嵌 ${count} 张本地图（${(bytes / 1048576).toFixed(2)} MB）`;
		};

		const importPack = async () => {
			const file = await pickPackFile();
			if (file === null) return null;
			const parsed = JSON.parse(await file.text());
			if (parsed === null || typeof parsed !== "object" || parsed.format !== PACK_FORMAT) throw new Error("不是 dsh-mini-skin 皮肤包");
			const images = parsed.images !== null && typeof parsed.images === "object" ? parsed.images : {};
			let restored = 0;
			for (const slot of ART_SLOTS) {
				const dataUrl = images[slot];
				if (typeof dataUrl !== "string" || dataUrl === "") continue;
				try {
					const blob = await dataUrlToBlob(dataUrl);
					await artStore(slot, blob);
					restored += 1;
				} catch {
					/* 单张失败不影响其余 */
				}
			}
			const modes = parsed.modes !== null && typeof parsed.modes === "object" ? parsed.modes : {};
			const next = buildSettings({
				...(parsed.shared !== null && typeof parsed.shared === "object" ? parsed.shared : {}),
				...modes,
				packName: typeof parsed.name === "string" ? parsed.name : "",
			});
			return { settings: next, message: `已导入「${next.packName === "" ? "未命名" : next.packName}」：恢复 ${restored} 张本地图` };
		};

		// ===================== 工作/空闲 检测 =====================
		// 宿主只在 DOM 上暴露状态（没有服务、没有事件）。判据按可靠性排序：
		//
		// 1) [data-chat-running] —— 会话级的"整轮在跑"标记（chat 视图的 RunningStatus，
		//    源码注释写明 "mount only while the Session is running"）。它覆盖**整轮**：
		//    思考、流式输出、执行工具、压缩上下文… 只要这一轮没结束就在。
		//    这是首选判据：之前只用下面两条，结果"工具执行/纯深度思考"时两者都不成立，
		//    被误判成空闲（实测反馈过的 bug）。
		// 2) [data-state='running'] —— 单个节点（推理行、工具卡、命令卡）自己还在流式时的标记。
		//    只在那个节点活着的时候有，且会被 settle 掉；作为 1) 的兜底（旧版宿主没有 1）。
		// 3) [data-composer-input][data-phase] —— 只有"提交中/裁决中"两个瞬时相位，
		//    同样只作兜底：它**不**代表整轮在跑。
		//
		// 原皮肤为此挂了整个 body 的 MutationObserver；本插件改用一个 0.4 秒定时器，每轮
		// 只做一两次 querySelector —— 代价是切换最多延迟 0.4 秒，收益是保住"0 观察器"。
		const WORK_POLL_MS = 400;
		let detectedWorking = false;
		let workOverride = null; // 面板在编辑某套预设时临时强制，关掉面板即恢复自动
		const isWorking = () => (workOverride === null ? detectedWorking : workOverride);
		const readWorking = () => {
			try {
				if (document.querySelector("[data-chat-running]") !== null) return true;
				if (document.querySelector("[data-state='running']") !== null) return true;
				const input = document.querySelector("[data-composer-input][data-phase]");
				const phase = input === null ? "" : input.getAttribute("data-phase");
				return phase === "submitting" || phase === "adjudicating";
			} catch {
				return false;
			}
		};

		// ===================== 皮肤库（插件自带目录，经节点半边读取）=====================
		const fetchLibrary = async () => {
			const response = await fetch(LIBRARY_ROUTE, { credentials: "same-origin" });
			if (!response.ok) throw new Error(`皮肤库读取失败（HTTP ${response.status}）`);
			const payload = await response.json();
			return { dir: typeof payload.dir === "string" ? payload.dir : "", skins: Array.isArray(payload.skins) ? payload.skins : [] };
		};

		const readLibrarySkin = async (file) => {
			const response = await fetch(`${LIBRARY_ROUTE}?file=${encodeURIComponent(file)}`, { credentials: "same-origin" });
			if (!response.ok) throw new Error(`读取「${file}」失败（HTTP ${response.status}）`);
			return await response.json();
		};

		const saveToLibrary = async (pack) => {
			const response = await fetch(LIBRARY_ROUTE, {
				method: "POST",
				credentials: "same-origin",
				headers: { "content-type": "application/json" },
				body: JSON.stringify({ pack }),
			});
			if (!response.ok) throw new Error(`存入皮肤库失败（HTTP ${response.status}）`);
			return await response.json();
		};

		/** 应用一个皮肤包：图片写回本机存储、设置套用、并记住指纹。 */
		const applySkinPack = async (pack, identity) => {
			const images = pack.images !== null && typeof pack.images === "object" ? pack.images : {};
			let restored = 0;
			for (const slot of ART_SLOTS) {
				const dataUrl = images[slot];
				if (typeof dataUrl !== "string" || dataUrl === "") continue;
				try {
					const blob = await dataUrlToBlob(dataUrl);
					await artStore(slot, blob);
					restored += 1;
				} catch {
					/* 单张失败不影响其余 */
				}
			}
			// v3 包是 dark:idle/dark:work/… 四键；v2 包只有 dark/light —— buildSettings 两种都认，
			// 所以这里必须原样展开，不能只挑 modes.dark / modes.light（否则 v3 包会整套回落成默认）。
			const modes = pack.modes !== null && typeof pack.modes === "object" ? pack.modes : {};
			const next = buildSettings({
				...(pack.shared !== null && typeof pack.shared === "object" ? pack.shared : {}),
				...modes,
				packName: typeof pack.name === "string" ? pack.name : "",
			});
			next.activeSkin = {
				kind: "pack",
				file: identity.file,
				name: next.packName === "" ? identity.file : next.packName,
				hash: fingerprint(next),
			};
			return { settings: next, restored };
		};

		/** 与拉条双向同步的数值框：输入过程中不提交，失焦或回车才夹到范围并写回。 */
		function NumberBox(props) {
			const [draft, setDraft] = React.useState(String(props.value));
			React.useEffect(() => {
				setDraft(String(props.value));
			}, [props.value]);
			const commit = (raw) => {
				// 清空输入框后失焦视为"放弃修改"，不要把它当成 0。
				if (String(raw).trim() === "") {
					setDraft(String(props.value));
					return;
				}
				const parsed = Number(raw);
				const next = Number.isFinite(parsed) ? Math.min(props.max, Math.max(props.min, Math.round(parsed))) : props.value;
				setDraft(String(next));
				props.onCommit(next);
			};
			return React.createElement(
				React.Fragment,
				null,
				React.createElement("input", {
					type: "number",
					min: props.min,
					max: props.max,
					step: 1,
					value: draft,
					"aria-label": props.label,
					style: NUMBER_STYLE,
					onChange: (event) => setDraft(event.currentTarget.value),
					onBlur: (event) => commit(event.currentTarget.value),
					onKeyDown: (event) => {
						if (event.key === "Enter") {
							commit(event.currentTarget.value);
							event.currentTarget.blur();
						}
					},
				}),
				props.unit === undefined || props.unit === null ? null : React.createElement("span", { style: UNIT_STYLE }, props.unit),
			);
		}

		/** 「自定义皮肤」分区。只有设置弹窗打开时才挂载；关掉即卸载，平时零成本。 */
		function CustomSkinPanel(props) {
			const [settings, setSettings] = React.useState(() => (typeof props.read === "function" ? props.read() : props.settings));
			// 提交放在事件处理器里，不放进 setState 的更新函数：更新函数里做副作用会被
			// React 重放或丢弃，正是"改了控件没反应"的常见来源。
			//
			// apply === false：只把值交给插件记账，**不要**立刻套用 —— 调用方（patchMode）
			// 已经为了"让这一套先生效"套用过一次了，套两次纯属浪费，还会和异步的重新套用打架。
			const change = (patch, apply = true) => {
				const next = { ...settings, ...patch };
				setSettings(next);
				edited.current = true;
				props.commit(next, apply);
			};

			// 面板一次编辑一套预设：默认跟随当前 DSH 主题，也可手动切去看另一套。
			const [editingMode, setEditingMode] = React.useState(() => presetKey(document.body !== null && document.body.hasAttribute("data-ds-dark-theme") ? "dark" : "light", isWorking()));
			// 本次面板生命周期里有没有真的改过值（ref：卸载清理函数里要读它，state 读不到最新值）。
			const edited = React.useRef(false);
			const m = settings[editingMode];
			const patchMode = (patch) => {
				// 改哪一套就让哪一套生效：setTheme 在偏好未变时会直接返回，所以拖滑块也不会重复切换。
				//
				// 这里只调 preview（纯套用，不落盘、不重新读皮肤包）：
				// 曾经这里调的是 reapply，而 reapply 在"当前皮肤来自皮肤库"时会重新读盘、
				// 把读回来的旧配置写回 settings 并落盘 —— 于是用户每改一次，过 0.几秒就被
				// 覆盖回当前皮肤的原配置。preview 没有这条写回路径。
				if (typeof props.switchMode === "function") props.switchMode(modeOf(editingMode));
				workOverride = editingMode.endsWith(":work");
				edited.current = true;
				if (typeof props.preview === "function") props.preview();
				change({ [editingMode]: { ...m, ...patch } }, false);
			};
			const [packStatus, setPackStatus] = React.useState("");
			const [library, setLibrary] = React.useState([]);
			const [libraryDir, setLibraryDir] = React.useState("");
			const [libraryError, setLibraryError] = React.useState("");

			const refreshLibrary = async () => {
				try {
					const payload = await fetchLibrary();
					setLibrary(payload.skins);
					setLibraryDir(payload.dir);
					setLibraryError("");
				} catch (error) {
					setLibrary([]);
					setLibraryError(error !== null && error !== undefined && error.message !== undefined ? error.message : String(error));
				}
			};
			React.useEffect(() => {
				void refreshLibrary();
				return () => {
					workOverride = null;
					// 只有本次面板真的被编辑过才重放：否则光开关一次设置面板，就会在
					// "当前皮肤来自皮肤库"时白白重新读一遍包（实测那个包有 17MB）。
					if (edited.current && typeof props.reapply === "function") props.reapply();
				};
			}, []);

			const currentHash = fingerprint(settings);
			const activeSkin = settings.activeSkin !== null && typeof settings.activeSkin === "object" ? settings.activeSkin : { kind: "custom" };
			const skinOnPack = activeSkin.kind !== "official" && typeof activeSkin.hash === "string" && activeSkin.hash === currentHash;
			// 官方皮肤是库里的一个普通包（official.json），但它的身份仍是"官方"：
			// 渲染时会把本插件的一切撤下（宿主原样），所以它显示成一个固定名字。
			const skinStatus = activeSkin.kind === "official" ? "官方皮肤（宿主原样）" : skinOnPack ? activeSkin.name : "自定义";
			const skinValue = activeSkin.kind === "official" ? "official" : skinOnPack ? `pack:${activeSkin.file}` : "custom";

			const applyLibrarySkin = async (value) => {
				if (value === "custom") return;
				if (value === "official") {
					// 官方皮肤的配置也来自那个包文件，所以这里是真的读一遍（不是写死常量）。
					setPackStatus("正在应用…");
					try {
						const pack = await readLibrarySkin(OFFICIAL_FILE);
						const next = buildSettings({
							...(pack.shared !== null && typeof pack.shared === "object" ? pack.shared : {}),
							...(pack.modes !== null && typeof pack.modes === "object" ? pack.modes : {}),
							packName: typeof pack.name === "string" ? pack.name : "",
						});
						next.activeSkin = { kind: "official", file: OFFICIAL_FILE, name: "官方皮肤（宿主原样）", hash: fingerprint(next) };
						setSettings(next);
						props.commit(next);
						setPackStatus("已切换到官方皮肤（宿主原样）。");
					} catch (error) {
						setPackStatus(`切换官方皮肤失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`);
					}
					return;
				}
				const file = value.startsWith("pack:") ? value.slice(5) : value;
				setPackStatus("正在应用…");
				try {
					const pack = await readLibrarySkin(file);
					const result = await applySkinPack(pack, { file });
					setSettings(result.settings);
					props.commit(result.settings);
					setPackStatus(`已应用「${result.settings.packName === "" ? file : result.settings.packName}」：恢复 ${result.restored} 张本地图`);
				} catch (error) {
					setPackStatus(`应用失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`);
				}
			};

			const runExport = async () => {
				setPackStatus("正在导出…");
				try {
					const message = await exportPack(settings);
					let extra = "";
					try {
						const built = await buildPack(settings);
						const saved = await saveToLibrary(built.pack);
						extra = `；已存入皮肤库（${saved.file}）`;
						await refreshLibrary();
					} catch (error) {
						extra = `；存入皮肤库失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`;
					}
					setPackStatus(message + extra);
				} catch (error) {
					setPackStatus(`导出失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`);
				}
			};

			const runImport = async () => {
				setPackStatus("正在导入…");
				try {
					const result = await importPack();
					if (result === null) {
						setPackStatus("已取消");
						return;
					}
					setSettings(result.settings);
					props.commit(result.settings);
					setPackStatus(result.message);
				} catch (error) {
					setPackStatus(`导入失败：${error !== null && error !== undefined && error.message !== undefined ? error.message : String(error)}`);
				}
			};

			// 选「本地文件…」时立刻弹系统选择器；用户取消就保持原选择不动。
			const AREA_FIELDS = { canvas: "canvasImage", sidebar: "sidebarImage", character: "characterImage" };
			const chooseAreaImage = async (area, value) => {
				const field = AREA_FIELDS[area];
				if (value !== "file") {
					patchMode({ [field]: value });
					return;
				}
				const name = await adoptLocalFile(`${area}:${editingMode}`);
				if (name === null) return;
				patchMode({ [field]: "file", [`${field}Name`]: name });
			};

			return React.createElement(
				"div",
				{ style: LIST_STYLE },
				React.createElement("div", { style: GROUP_TITLE_STYLE }, "应用已有皮肤"),
				Row(
					"正在使用",
					libraryError === "" ? `皮肤库：${library.filter((row) => row.ok && row.file !== OFFICIAL_FILE).length} 个可用${library.filter((row) => !row.ok).length > 0 ? `，${library.filter((row) => !row.ok).length} 个无法解析` : ""}` : libraryError,
					Select(
						skinValue,
						[
							{ value: "official", label: "官方皮肤（宿主原样）" },
							// official.json 不进这个列表：上面那一项就是它，否则会重复出现两条官方皮肤。
							...library.filter((row) => row.file !== OFFICIAL_FILE).map((row) => ({ value: `pack:${row.file}`, label: row.ok ? `皮肤：${row.name}` : `⚠ ${row.file}（${row.error}）`, disabled: !row.ok })),
							...(skinValue === "custom" ? [{ value: "custom", label: "自定义（当前设置）" }] : []),
						],
						(value) => void applyLibrarySkin(value),
					),
				),
				Row(
					"皮肤库目录",
					libraryDir === "" ? "插件自己的目录，把导出的 .json 放进这里就会被读到。" : libraryDir,
					React.createElement("button", { type: "button", style: { ...SELECT_STYLE, cursor: "pointer" }, onClick: () => void refreshLibrary() }, "重新读取"),
				),
				packStatus === "" ? null : Row("状态", packStatus, null),
				Row(
					"正在编辑",
					"深色/浅色 × 空闲/工作 = 四套预设。切换这里会同时把界面主题与预览状态切过去；关掉面板后按真实工作状态自动切换。",
					Select(editingMode, [{ value: "dark:idle", label: "深色 · 空闲" }, { value: "dark:work", label: "深色 · 工作" }, { value: "light:idle", label: "浅色 · 空闲" }, { value: "light:work", label: "浅色 · 工作" }], (value) => {
						setEditingMode(value);
						workOverride = value.endsWith(":work");
						if (typeof props.switchMode === "function") props.switchMode(modeOf(value));
						if (typeof props.reapply === "function") props.reapply();
					}),

				Row(

				),
				m.canvasImage === "none" ? null : React.createElement(React.Fragment, null,
				Row(
					"主内容 · 背景图",
					m.canvasFileName !== "" ? `已选择本地文件：${m.canvasFileName}` : m.canvasImage === "file" ? "尚未选择图片。" : `当前是内置素材（${m.canvasImage}），选一张本地图片即可替换。`,
					Select(m.canvasImage, materialChoices(CANVAS_CHOICES, m.canvasImage), (value) => chooseAreaImage("canvas", value)),

		// ===================== 装载 =====================
		function apply(ctx) {
			const claimant = Symbol("dsh-mini-skin-activation");
			if (!activate(claimant)) document.addEventListener("DOMContentLoaded", () => activate(claimant), { once: true });

			// 诊断用：把每一次"真正写进去"的图片变量记下来。
			// 排查"切空闲/工作态时闪一下"这类问题时，先 __dshMiniSkinTrace(true) 再切一次状态，
			// 然后看 __dshMiniSkinTrace() 的输出 —— 若里面出现了某条记录，说明那一层确实被重写了
			// （重写 = 重新解码 = 一帧空白）；若一条都没有，问题就不在这次重放里。
			// 注意：这里必须替换**模块级**的 writeArtVar（applySettings 闭包引用的是它），
			// 不能在 apply 里再声明一个同名的局部变量把它遮蔽掉。
			const traceRecords = [];
			window.__dshMiniSkinTrace = (on) => {
				if (on !== true) return traceRecords.slice();
				traceRecords.length = 0;
				const original = writeArtVar;
				writeArtVar = (name, value) => {
					if (lastArtWritten.get(name) !== value) traceRecords.push({ name, value: String(value).slice(0, 60), at: Math.round(performance.now()) });
					original(name, value);
				};
				return traceRecords;
			};

			let settings = readSettings();
			detectedWorking = readWorking();
			applySettings(claimant, settings);

			// 当前皮肤来自皮肤库时，用磁盘上的包重新推导一次设置在启动时也要做 ——
			// 否则「官方皮肤」的配置就会退回代码里的默认值（它现在是一份包文件，不是常量）；
			// 顺便也让"在磁盘上改过包、刷新页面"能生效。落地前先比指纹，没变就一个字节都不动。
			const hydrateFromLibrary = async () => {
				try {
					const skin = settings.activeSkin;
					if (skin === null || typeof skin !== "object") return;
					const official = skin.kind === "official";
					let file = official ? OFFICIAL_FILE : skin.kind === "pack" && typeof skin.file === "string" && skin.file !== "" ? skin.file : "";
					if (file === "") return;
					let pack = null;
					try {
						pack = await readLibrarySkin(file);
					} catch {
						// 这个文件不在了（被删或被改名）：按包名在库里找同名的那份接手，
						// 让活动皮肤自己认得回来，而不是卡在一个不存在的文件上。
						if (official) throw new Error("官方皮肤包缺失");
						const wanted = typeof skin.name === "string" ? skin.name : "";
						const rows = wanted === "" ? [] : (await fetchLibrary()).skins;
						const replacement = rows.find((row) => row.ok === true && row.name === wanted && row.file !== file);
						if (replacement === undefined) throw new Error("皮肤包不存在");
						file = replacement.file;
						pack = await readLibrarySkin(file);
					}
					const next = official
						? buildSettings({
							...(pack.shared !== null && typeof pack.shared === "object" ? pack.shared : {}),
							...(pack.modes !== null && typeof pack.modes === "object" ? pack.modes : {}),
							packName: typeof pack.name === "string" ? pack.name : "",
						})
						: (await applySkinPack(pack, { file })).settings;
					next.activeSkin = official
						? { kind: "official", file: OFFICIAL_FILE, name: "官方皮肤（宿主原样）", hash: fingerprint(next) }
						: { ...skin, file, name: next.packName === "" ? file : next.packName, hash: fingerprint(next) };
					if (fingerprint(next) === fingerprint(settings) && next.activeSkin.file === skin.file) return;
					settings = next;
					persistSettings(next);
					if (document.body !== null) applySettings(claimant, next);
				} catch {
					/* 皮肤包读不到（被删了 / 路由还没起来）：保持当前设置 */
				}
			};
			void hydrateFromLibrary();

			// 状态轮询：只在状态真的变了时重放设置；面板编辑期间由 workOverride 接管。
			// 整段包在 try 里：这是定时器，一旦抛出且没人接住，就会按 0.4 秒的节奏刷屏并让
			// 皮肤状态卡死在出错的那一拍。
			const poll = () => {
				try {
					const now = readWorking();
					if (now === detectedWorking) return;
					detectedWorking = now;
					if (workOverride === null && document.body !== null) applySettings(claimant, settings);
				} catch {
					/* 轮询失败不影响这一轮之外的东西 */
				}
			};
			const workTimer = window.setInterval(poll, WORK_POLL_MS);
			ctx.effect(() => () => window.clearInterval(workTimer), "dsh-mini-skin: work-state poll");

			// 启动时预热本地图片的 blob URL（存在 IndexedDB 里），再重放一次设置；
			// 之后就交给 artUrl 的惰性缓存，用到哪张取哪张。
			//
			// 顺序：先按已落地的设置把画面铺好（上面那次 applySettings），再读皮肤包推导
			// 一次（hydrateFromLibrary）。包里引用的本地图会被 applySkinPack 写回存储并进
			// 缓存，所以它结束时可以直接再套一遍。
			void (async () => {
				for (const slot of ART_SLOTS) {
					try {
						await artUrl(slot);
					} catch {
						/* 读不到就当作没有本地图 */
					}
				}
				if (document.body !== null) applySettings(claimant, settings);
			})();

			// 切浅/深色时换成对应预设。色值取自事件快照（active.colorScheme），绝不读 DOM；
			// 订阅走 ctx.inject，避免 apply 早于 theme 服务注册时静默漏订阅。
			const resync = (snapshot) => {
				const active = snapshot !== null && typeof snapshot === "object" ? snapshot.active : undefined;
				const scheme = active !== null && typeof active === "object" ? active.colorScheme : undefined;
				if (scheme === "dark" || scheme === "light") knownScheme = scheme;
				applySettings(claimant, settings);
			};
			const subscribeTheme = (themeCtx) => {
				const theme = typeof themeCtx.get === "function" ? themeCtx.get("theme") : undefined;
				if (theme === undefined || typeof themeCtx.on !== "function") return;
				themeCtx.on("theme/change", resync);
				themeCtx.effect(() => () => themeCtx.off("theme/change", resync), "dsh-mini-skin: theme resync");
			};
			if (typeof ctx.inject === "function") ctx.inject(["theme"], subscribeTheme);
			else subscribeTheme(ctx);

			// 设置分区：用 ctx.inject 等 slots 服务就绪再注册（本插件 dsh.client.inject 为空，
			// apply 可能早于 slots 注册，查一次 ctx.get 会拿到 undefined）。
			const registerSection = (slotsCtx) => {
				const slots = typeof slotsCtx.get === "function" ? slotsCtx.get("slots") : undefined;
				if (slots === undefined || typeof slots.inject !== "function") return;
				slotsCtx.effect(
					() =>
						slots.inject("settings.section", () =>
							slots.register(
								{
									name: "settings.section",
									id: SECTION_ID,
									order: 116,
									label: () => SECTION_TITLE,
									inject: () => ({
											// 面板每次挂载都通过它取当前值：不能传对象快照，否则重新挂载后面板会显示旧值
											// （曾经因此出现"下拉显示旧选项、再选同一项因值未变而不触发"的问题）。
											read: () => settings,
										settings,
										commit: (next, apply = true) => {
											settings = next;
											persistSettings(next);
											if (apply !== false && document.body !== null) applySettings(claimant, next);
										},
										// 纯套用：只把当前内存里的设置画到页面上，不落盘、不重新读皮肤包。
										// 面板改一个值时用它做"立刻生效"，绝不会覆盖用户刚改的值。
										preview: () => {
											if (document.body !== null) applySettings(claimant, settings);
										},
										reapply: () => {
										// 若当前用的皮肤来自皮肤库，重新读一遍再套用 —— 否则磁盘上更新过的包点了也没反应。
										//
										// 关键：这条路径**只影响页面**，绝不写回 settings / localStorage。
										// 一旦写回，用户在面板里刚改的值会被包里的旧值覆盖（实测反馈过的 bug）。
										const skin = settings.activeSkin;
										const fromLibrary = skin !== null && typeof skin === "object" && skin.kind === "pack" && typeof skin.file === "string" && skin.file !== "";
										if (!fromLibrary) { if (document.body !== null) applySettings(claimant, settings); return Promise.resolve(); }
										return readLibrarySkin(skin.file).then((pack) => applySkinPack(pack, { file: skin.file })).then((result) => {
										if (document.body !== null) applySettings(claimant, result.settings);
										}).catch(() => { if (document.body !== null) applySettings(claimant, settings); });
										},
										switchMode: (mode) => {
											// 宿主的唯一偏好写入口：真的切主题（并持久化），这样配另一套预设时能立刻看到效果。
											// 先记账再切换：setTheme 之后宿主要到下一帧才把主题属性写进 DOM。
											if (mode === "dark" || mode === "light") knownScheme = mode;
											const theme = typeof ctx.get === "function" ? ctx.get("theme") : undefined;
											if (theme !== undefined && typeof theme.setTheme === "function") theme.setTheme(mode);
										},
									}),
								},
								CustomSkinPanel,
							),
						),
					"dsh-mini-skin: settings section",
				);
			};
			if (typeof ctx.inject === "function") ctx.inject(["slots"], registerSection);
			else registerSection(ctx);

			ctx.effect(
				() => () => {
					clearSettings(claimant);
					deactivate(claimant);
				},
				"dsh-mini-skin: stylesheet",
			);
		}

		exports.apply = apply;
		// 模块级 inject：客户端运行器会等 slots 服务就绪再调用 apply（皮肤管理器同款做法）。
		// apply 里仍保留 ctx.inject 兜底，两条路径都通。
		exports.inject = ["slots"];
		return module.exports;
	},
});
